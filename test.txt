NUMB pBoton;
        pBoton = 2;
        NUMB pLed;
        pLed = 8;
        
        CONF Pin pLed OUT;
        CONF Pin pBoton IN;
        
        LOOP {
            NUMB estadoBoton;
            estadoBoton = DREAD(pBoton);
            WHEN (estadoBoton == HIGH) THEN {
                PIN pLed ON;
            };ELSE {
                PIN pLed OFF;
            };
        };
